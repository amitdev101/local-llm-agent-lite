import java.awt.event.*;

public class SnakeGame extends JPanel implements KeyListener {
    private int snakeX = 100, snakeY = 100;
    private int direction = 0; // 0-right, 1-up, 2-left, 3-down

    private boolean gameOver = false;

    public SnakeGame() {
        new Food();
        setFocusable(true);
        setPreferredSize(new Dimension(400, 400));
        addKeyListener(this);
    }

    public void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.setColor(Color.GREEN);
        g.fillOval(snakeX, snakeY, 10, 10);
        g.setColor(Color.RED);
        g.fillOval(foodX, foodY, 10, 10);
    }

    public void move() {
        switch (direction) {
            case 0: snakeX += 10; break;
            case 1: snakeY -= 10; break;
            case 2: snakeX -= 10; break;
            case 3: snakeY += 10; break;
        }
        if (snakeX < 0 || snakeX >= 400 || snakeY < ight || snakeY >= 400) {
            gameOver = true;
        }
        if (snakeX == foodX && snakeY == foodY) {
            // Grow snake
            snakeX = 100;
            snakeY = 100;
            new Food();
        }
    }

    public void changeDirection(int newDirection) {
        direction = newDirection;
    }

    private class Food {
        public Food() {
            foodX = (int) (Math.random() * 40);
            foodY = (int) (Math.random() * 40);
        }
    }

    @Override
    public void keyPressed(KeyEvent e) {
        int key = e.getKeyCode();
        if (key == KeyEvent.VK_LEFT) {
            changeDirection(2);

            changeDirection(0);
        } else if (key == KeyEvent.VK_UP) {
            changeDirection(1);
        } else if (key == KeyEvent.VK_DOWN) {
            changeDirection(3);
        }
    }

    @Override
    public void keyTyped(KeyEvent e) {}
