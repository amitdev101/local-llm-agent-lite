    private Timer timer;
    private int snakeX = 100, snakeY = 100;
    private int directionX = 0, directionY = 0;
    private boolean running = false;
    private int foodX, foodY;

    public SnakeGame() {
        newFood();
        timer = new Timer(100, this);
        timer.start();
        addKeyListener(new KeyAdapter() {
            public void keyPressed(KeyEvent e) {
                handleKeyPress(e);
            }
        });
        setFocusable(true);
    }

    private void newFood() {
        foodX = (int) (Math.random() * 500);
        foodY = (int) (Math.random() * 500);
    }

    private void handleKeyPress(KeyEvent e) {
        int key = e.getKeyCode();
        if (key == KeyEvent.VK_UP) {
            directionX = 0;
            directionY = -10;
        } else if (key == KeyEvent.VK_DOWN) {
            directionX = 0;
            directionY = 10;
        } else if (key == KeyEvent.VK_LEFT) {
            directionX = -10;
            directionY = 0;
        } else if (key == KeyEvent.VK_RIGHT) {
            directionX = 10;
            directionY = 0;
        }
    }

    public void actionPerformed(ActionEvent e) {
        moveSnake();
        repaint();
    }

    private void moveSnake() {
        snakeX += directionX;
        snakeY += directionY;
        if (snakeX < 0 || snakeX >= 500 || snakeY < 0 || snakeY >= 500) {
            running = false;
        }
        if (snakeX == foodX && snakeY == foodY) {
            newFood();
        }
    }

    @Override
    public void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.setColor(Color.GREEN);
        g.fillRect(snakeX, snakeY, 10, 10);

        g.fillRect(foodX, foodY, 10, 10);
    }
