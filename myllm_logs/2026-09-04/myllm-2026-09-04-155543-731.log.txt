import javax.swing.*;
import java.awt.*;
import java.awt.event.*;

public class SnakeGame extends JPanel implements KeyListener {
    private int snakeX = 100, snakeY = 100;
    private int directionX = 0, directionY = 0;
    private int foodX, foodY;
    private boolean running = false;

    public SnakeGame() {
        newFood();
        running = true;
        setFocusable(true);
        setFocusTraversalKeysEnabled(false);
    }

    private void newFood() {
        foodX = (int) (Math.random() * 400);
        foodY = (int) (Math.random() * 400);
    }

    @Override
    public void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.setColor(Color.RED);
        g.fillOval(snakeX, snakeY, 10, 10);
        g.setColor(Color.GREEN);
        g.fillOval(foodX, foodY, 10, 10);
    }

    @Override
    public void keyPressed(KeyEvent e) {
        int key = e.getKeyCode();
        if (key == KeyEvent.VK_UP) {
            directionX = 0;
            directionY = -10;
        } else if (key == KeyEvent.VK_DOWN) {
            directionX = 0;
            directionY = 10;
        } else if (key == KeyEvent.VK_LEFT) {
            directionX = -10;
            directionY = 0;
        } else if (key == KeyEvent.VK_RIGHT) {
            directionX = 10;
            directionY = 0;
        }
    }

    @Override
    public void keyReleased(KeyEvent e) {}

    @Override
    public void keyTyped(KeyEvent e) {}

    public void run() {
        while (running) {
            moveSnake();
            repaint();
            try {
                Thread.sleep(100);
            } catch (InterruptedException ex) {
                ex.printStackTrace();
            }
        }
    }

    private void moveSnake() {
        snakeX += directionX;
        snakeY += directionY;

        // Check for collision with walls
        if (snakeX < 0 || snakeX > 400 || snakeY < 0 || snakeY > 400) {
            running = false;
        }

        // Check for collision with food
        if (snakeX == foodX && snakeY == foodY) {
            newFood();
        }
    }

    public static void main(String[] args) {
        JFrame frame = new JFrame("Snake Game");
        SnakeGame game = new SnakeGame();
        frame.add(game);
        frame.setSize(500, 500);
        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        frame.setVisible(true);
        new Thread(game).start();
    }
}

import javax.swing.*;
import java.awt.*;
import java.awt.event.*;
import java.util.*;

public class SnakeGame extends JPanel implements KeyListener {
    private static final int WIDTH = 600;
    private static final int HEIGHT = 600;
    private static final int SNAKE_SIZE = 20;
    private static final int FOOD_SIZE = 20;
    private static final int DELAY = 100;

    private int[][] snake;
    private int foodX, foodY;
    private Direction direction;
    private boolean gameOver;

    public SnakeGame() {
        setPreferredSize(new Dimension(WIDTH, HEIGHT));
        setBackground(Color.BLACK);
        setFocusable(true);
        setFocusTraversalKeysEnabled(false);
        
        // Initialize snake
        snake = new int[1][2];
        snake[0][0] = WIDTH / 2;
        snake[0][1] = HEIGHT / 2;
        
        // Initialize food
        foodX = (int) (Math.random() * (WIDTH - FOOD_SIZE));
        foodY = (int) (Math.random() * (HEIGHT - FOOD_SIZE));
        
        // Initialize direction
        direction = Direction.RIGHT;
        
        // Start game loop
        new Timer(DELAY, e -> {
            if (!gameOver) {
                moveSnake();
                repaint();
            }
        }).start();
    }

    private void moveSnake() {
        int headX = snake[0][0];
        int headY = snake[0][1];
        
        switch (direction) {
            case UP:
                headY -= SNAKE_SIZE;
                break;
            case DOWN:
                headY += SNAKE_SIZE;
                break;
            case LEFT:
                headX -= SNAKE_SIZE;
                break;
            case RIGHT:
                headX += SNAKE_SIZE;
                break;
        }
        
        // Check for collision with walls
        if (headX < 0 || headX >= WIDTH || headY < 0 || headY >= HEIGHT) {
            gameOver = true;
            return;
        }
        
        // Check for collision with self
        for (int i = 1; i < snake.length; i++) {
            if (snake[i][0] == headX && snake[i][1] == headY) {
                gameOver = true;
                return;
            }
        }
        
        // Check for collision with food
        if (headX == foodX && headY == foodY) {
            // Grow the snake
            int[] newHead = {headX, headY};
            int[][] newSnake = new int[snake.length + 1][2];
            
            for (int i = 0; i < snake.length; i++) {
                newSnake[i][0] = snake[i][0];
                newSnake[i][1] = snake[i][1];
            }
            
            newSnake[snake.length][0] = headX;
            newSnake[snake.length][1] = headY;
            
            snake = newSnake;
            
            // Generate new food
            foodX = (int) (Math.random() * (WIDTH - FOOD_SIZE));
            foodY = (int) (Math.random() * (HEIGHT - FOOD_SIZE));
        } else {
            // Move the snake
            int[] newHead = {headX, headY};
            int[][] newSnake = new int[snake.length][2];
            
            for (int i = 1; i < snake.length; i++) {
                newSnake[i][0] = snake[i - 1][0];
                newSnake[i][1] = snake[i - 1][1];
            }
            
            newSnake[0][0] = headX;
            newSnake[0][1] = headY;
            
            snake = newSnake;
        }
    }

    @Override
    public void paintComponent(Graphics g) {
        super.paintComponent(g);
        
        // Draw food
        g.setColor(Color.RED);
        g.fillOval(foodX, foodY, FOOD_SIZE, FOOD_SIZE);
        
        // Draw snake
        g.setColor(Color.GREEN);
        for (int[] segment : snake) {
            g.fillOval(segment[0], segment[1], SNAKE_SIZE, SNAKE_SIZE);
        }
    }

    @Override
    public void keyPressed(KeyEvent e) {
        int code = e.getKeyCode();
        
        if (code == KeyEvent.VK_UP && direction != Direction.DOWN) {
            direction = Direction.UP;
        } else if (code == KeyEvent.VK_DOWN && direction != Direction.UP) {
            direction = Direction.DOWN;
        } else if (code == KeyEvent.VK_LEFT && direction != Direction.RIGHT) {
            direction = Direction.LEFT;
        } else if (code == KeyEvent.VK_RIGHT && direction != Direction.LEFT) {
            direction = Direction.RIGHT;
        }
    }

    @Override
    public void keyReleased(KeyEvent e) {}

    @Override
    public void keyTyped(KeyEvent e) {}

    private enum Direction {
        UP, DOWN, LEFT, RIGHT
    }
}

public class GameFrame extends JFrame {
    public GameFrame() {
        add(new SnakeGame());
        pack();
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);
        setVisible(true);
    }
}

public class SnakeGameApp {
    public static void main(String[] args) {
        new GameFrame();
    }
}
